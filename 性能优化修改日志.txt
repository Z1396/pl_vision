```cpp

    frame_intervals.erase(frame_intervals.begin());  // O(n) 操作
}
double avg_interval = 0.0;
for (double interval : frame_intervals) {  // O(n) 遍历
    avg_interval += interval;

int frame_idx = 0;
double interval_sum = 0.0;

interval_sum -= frame_intervals[frame_idx];      // O(1)
interval_sum += frame_interval;                   // O(1)
frame_intervals[frame_idx] = frame_interval;
frame_idx = (frame_idx + 1) % FPS_SMOOTH_WINDOW;  // O(1)

    while (running_) {
        BroadcastMessage msg;
        {
            std::unique_lock<std::mutex> lock(queue_mutex_);
            queue_cv_.wait(lock, ...);
            msg = broadcast_queue_.front();
            broadcast_queue_.pop();
        }
        // 发送 msg
    }
}

// 修改后：双缓冲读取
void WebSocketServer::broadcastLoop() {
    while (running_) {
        std::string video_frame, data_frame;
        bool send_video = false, send_data = false;
        
        {
            std::unique_lock<std::mutex> lock(video_mutex_);
            frame_cv_.wait_for(lock, std::chrono::milliseconds(10), ...);
            
            if (has_new_video_) {
                video_frame = latest_video_frame_;  // 直接复制最新帧
                has_new_video_ = false;
                send_video = true;
            }
        }
        // 同样处理 data_frame
        // 发送最新帧
    }
}
```

    std::lock_guard<std::mutex> lock(video_mutex_);
